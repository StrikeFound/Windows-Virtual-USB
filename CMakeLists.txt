cmake_minimum_required(VERSION 3.24)
project(WindowsVirtualUsb VERSION 0.1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE)
add_link_options(/PDBALTPATH:%_PDB%)
add_library(VirtualUsbSDK STATIC VirtualUsb/VirtualUsb.cpp)
add_library(VirtualUsb::SDK ALIAS VirtualUsbSDK)
target_include_directories(VirtualUsbSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/VirtualUsb")
set_target_properties(VirtualUsbSDK PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/SDK")
add_executable(WindowsVirtualUsbSetupNative InstallerConsole/Native/Setup.cpp)
target_link_libraries(WindowsVirtualUsbSetupNative PRIVATE setupapi newdev cfgmgr32 advapi32 wintrust crypt32)
target_link_options(WindowsVirtualUsbSetupNative PRIVATE /MANIFESTUAC:level='requireAdministrator')
set_target_properties(WindowsVirtualUsbSetupNative PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Native")
add_library(WindowsVirtualUsbBridge SHARED ExamplesUI/NativeBridge.cpp)
target_link_libraries(WindowsVirtualUsbBridge PRIVATE VirtualUsb::SDK)
set_target_properties(WindowsVirtualUsbBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/ExamplesUI")
foreach(name Keyboard Mouse Xbox360 DualShock4 XboxOneGip)
  add_executable("WindowsVirtualUsb${name}UI" WIN32 "ExamplesUI/${name}.cpp")
  target_link_libraries("WindowsVirtualUsb${name}UI" PRIVATE VirtualUsb::SDK advapi32 user32 gdi32 comctl32 shell32)
  set_target_properties("WindowsVirtualUsb${name}UI" PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/ExamplesUI")
endforeach()
enable_testing()
add_executable(ExampleTests Tests/ExampleTests.cpp)
target_link_libraries(ExampleTests PRIVATE VirtualUsb::SDK)
set_target_properties(ExampleTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
add_test(NAME ExampleProtocols COMMAND ExampleTests)
add_executable(EndpointTests Tests/EndpointTests.cpp)
target_link_libraries(EndpointTests PRIVATE VirtualUsb::SDK)
set_target_properties(EndpointTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
add_test(NAME SoleInputEndpoint COMMAND EndpointTests)
add_executable(InstallerNativeTests Tests/InstallerNativeTests.cpp)
target_link_libraries(InstallerNativeTests PRIVATE setupapi newdev cfgmgr32 advapi32 wintrust crypt32)
set_target_properties(InstallerNativeTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
add_test(NAME InstallerNative COMMAND InstallerNativeTests "${CMAKE_CURRENT_SOURCE_DIR}/InstallerUI/Package/WindowsVirtualUsb.inf")
add_executable(ExampleLiveTests Tests/ExampleLiveTests.cpp)
target_link_libraries(ExampleLiveTests PRIVATE VirtualUsb::SDK advapi32 user32 setupapi hid xinput)
set_target_properties(ExampleLiveTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
# The live test is opt-in inside a Hyper-V guest, never an automatic CTest.
